51std::string make_xml_id(std::string_view prefix) {
52 static thread_local std::mt19937_64 rng(
53 static_cast<std::uint64_t>(
54 std::chrono::steady_clock::now().time_since_epoch().count()));
55 std::uniform_int_distribution<std::uint64_t> dist;
56 std::ostringstream oss;
57 oss << prefix << '-' << std::hex << dist(rng) << dist(rng);
58 return oss.str();
59}

64std::string utc_iso8601_now() {
65 const auto now = std::chrono::system_clock::now();
66 const auto tt = std::chrono::system_clock::to_time_t(now);
67 std::tm tm{};
68#ifdef _WIN32
69 gmtime_s(&tm, &tt);
70#else
71 gmtime_r(&tt, &tm);
72#endif
73 char buf[32];
74 std::strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &tm);
75 return std::string(buf);
76}

106void add_registry_package(pugi::xml_node leaf_list, const submission_set& s) {
107 auto rp = leaf_list.append_child("rim:RegistryPackage");
108 rp.append_attribute("id") = s.entry_uuid.c_str();
109 rp.append_attribute("objectType") =
110 "urn:oasis:names:tc:ebxml-regrep:ObjectType:RegistryObject:RegistryPackage";
111
112 if (!s.author_person.empty() || !s.author_institution.empty()) {
113 auto cls = rp.append_child("rim:Classification");
114 cls.append_attribute("classificationScheme") =
115 "urn:uuid:a7058bb9-b4e4-4307-ba5b-e3f0ab85e12d";
116 cls.append_attribute("classifiedObject") = s.entry_uuid.c_str();
117 cls.append_attribute("nodeRepresentation") = "";
118 cls.append_attribute("id") = make_xml_id("cls").c_str();
119 cls.append_attribute("objectType") =
120 "urn:oasis:names:tc:ebxml-regrep:ObjectType:RegistryObject:Classification";
121 if (!s.author_person.empty()) {
122 auto slot = cls.append_child("rim:Slot");
123 slot.append_attribute("name") = "authorPerson";
124 auto vl = slot.append_child("rim:ValueList");
125 vl.append_child("rim:Value")
126 .text()
127 .set(s.author_person.c_str());
128 }
129 if (!s.author_institution.empty()) {
130 auto slot = cls.append_child("rim:Slot");
131 slot.append_attribute("name") = "authorInstitution";
132 auto vl = slot.append_child("rim:ValueList");
133 vl.append_child("rim:Value")
134 .text()
135 .set(s.author_institution.c_str());
136 }
137 }
138
139 if (!s.submission_time.empty()) {
140 auto slot = rp.append_child("rim:Slot");
141 slot.append_attribute("name") = "submissionTime";
142 auto vl = slot.append_child("rim:ValueList");
143 vl.append_child("rim:Value").text().set(s.submission_time.c_str());
144 }
145
146 add_classification(rp, kSubmissionSetClassification,
147 s.entry_uuid.c_str(), "", "ssc");
148
149 add_external_identifier(rp, kSubmissionSetIdScheme, s.entry_uuid.c_str(),
150 s.submission_set_unique_id.c_str(),
151 "XDSSubmissionSet.uniqueId");
152 add_external_identifier(rp, kSubmissionSetIdScheme, s.entry_uuid.c_str(),
153 s.source_id.c_str(), "XDSSubmissionSet.sourceId");
154 add_external_identifier(rp, kSubmissionSetIdScheme, s.entry_uuid.c_str(),
155 s.patient_id.c_str(), "XDSSubmissionSet.patientId");
156}

201kcenon::common::Result<built_envelope> build_iti41_envelope(
202 const submission_set& s, const std::vector<std::string>& cid_refs) {
203 if (cid_refs.size() != s.documents.size()) {
204 return kcenon::common::make_error<built_envelope>(
205 static_cast<int>(error_code::envelope_build_failed),
206 "content-id list size does not match document count",
207 std::string(error_source));
208 }
209
210 pugi::xml_document doc;
211 auto decl = doc.append_child(pugi::node_declaration);
212 decl.append_attribute("version") = "1.0";
213 decl.append_attribute("encoding") = "UTF-8";
214
215 auto env = doc.append_child("soap:Envelope");
216 env.append_attribute("xmlns:soap") = kSoapEnv;
217 env.append_attribute("xmlns:wsa") = kWsa;
218 env.append_attribute("xmlns:wsse") = kWsse;
219 env.append_attribute("xmlns:wsu") = kWsu;
220 env.append_attribute("xmlns:lcm") = kLcm;
221 env.append_attribute("xmlns:rim") = kRim;
222 env.append_attribute("xmlns:xop") = kXop;
223
224 auto header = env.append_child("soap:Header");
225
226 auto action = header.append_child("wsa:Action");
227 action.append_attribute("soap:mustUnderstand") = "true";
228 action.text().set("urn:ihe:iti:2007:RegisterDocumentSet-b");
229
230 auto message_id = header.append_child("wsa:MessageID");
231 message_id.text().set(("urn:uuid:" + make_xml_id("msg")).c_str());
232
233 auto security = header.append_child("wsse:Security");
234 security.append_attribute("soap:mustUnderstand") = "true";
235
236 built_envelope out{};
237 out.timestamp_id = make_xml_id("ts");
238 out.binary_security_token_id = make_xml_id("bst");
239 out.body_id = make_xml_id("body");
240
241 auto timestamp = security.append_child("wsu:Timestamp");
242 timestamp.append_attribute("wsu:Id") = out.timestamp_id.c_str();
243 auto created = timestamp.append_child("wsu:Created");
244 const std::string now = utc_iso8601_now();
245 created.text().set(now.c_str());
246
247 auto bst = security.append_child("wsse:BinarySecurityToken");
248 bst.append_attribute("wsu:Id") = out.binary_security_token_id.c_str();
249 bst.append_attribute("EncodingType") =
250 "http://docs.oasis-open.org/wss/2004/01/oasis-200401-wss-soap-message-"
251 "security-1.0#Base64Binary";
252 bst.append_attribute("ValueType") =
253 "http://docs.oasis-open.org/wss/2004/01/oasis-200401-wss-x509-token-"
254 "profile-1.0#X509v3";
255
256 auto body = env.append_child("soap:Body");
257 body.append_attribute("wsu:Id") = out.body_id.c_str();
258
259 auto req = body.append_child("lcm:SubmitObjectsRequest");
260
261 auto leaf_list = req.append_child("rim:RegistryObjectList");
262
263 if (s.patient_id.empty()) {
264 return kcenon::common::make_error<built_envelope>(
265 static_cast<int>(error_code::metadata_missing_patient_id),
266 "submission_set.patient_id is empty",
267 std::string(error_source));
268 }
269 if (s.source_id.empty()) {
270 return kcenon::common::make_error<built_envelope>(
271 static_cast<int>(error_code::metadata_missing_source_id),
272 "submission_set.source_id is empty",
273 std::string(error_source));
274 }
275 if (s.submission_set_unique_id.empty()) {
276 return kcenon::common::make_error<built_envelope>(
277 static_cast<int>(error_code::metadata_missing_submission_set_uid),
278 "submission_set.submission_set_unique_id is empty",
279 std::string(error_source));
280 }
281 if (s.documents.empty()) {
282 return kcenon::common::make_error<built_envelope>(
283 static_cast<int>(error_code::metadata_no_documents),
284 "submission_set contains no documents",
285 std::string(error_source));
286 }
287
288 for (std::size_t i = 0; i < s.documents.size(); ++i) {
289 const auto& d = s.documents[i];
290 if (d.unique_id.empty()) {
291 return kcenon::common::make_error<built_envelope>(
292 static_cast<int>(
293 error_code::metadata_document_missing_unique_id),
294 "document[" + std::to_string(i) + "].unique_id is empty",
295 std::string(error_source));
296 }
297 if (d.mime_type.empty()) {
298 return kcenon::common::make_error<built_envelope>(
299 static_cast<int>(
300 error_code::metadata_document_missing_mime_type),
301 "document[" + std::to_string(i) + "].mime_type is empty",
302 std::string(error_source));
303 }
304 if (d.content.empty()) {
305 return kcenon::common::make_error<built_envelope>(
306 static_cast<int>(error_code::metadata_document_empty),
307 "document[" + std::to_string(i) + "].content is empty",
308 std::string(error_source));
309 }
310 add_document_entry(leaf_list, d, cid_refs[i]);
311 }
312
313 add_registry_package(leaf_list, s);
314
315 xml_string_writer writer;
316 doc.save(writer, "", pugi::format_raw | pugi::format_no_declaration,
317 pugi::encoding_utf8);
318 if (writer.out.empty()) {
319 return kcenon::common::make_error<built_envelope>(
320 static_cast<int>(error_code::envelope_serialization_failed),
321 "pugixml produced an empty envelope",
322 std::string(error_source));
323 }
324
325 out.xml = R"(<?xml version="1.0" encoding="UTF-8"?>)";
326 out.xml += writer.out;
327 return out;
328}
