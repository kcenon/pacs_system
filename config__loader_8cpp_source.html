71private:
72 void parse(std::string_view content) {
73 std::istringstream stream{std::string(content)};
74 std::string line;
75 std::vector<std::pair<int, std::string>> path_stack;
76 std::string current_list_path;
77
78 while (std::getline(stream, line)) {
79 // Skip empty lines
80 if (line.empty() || line.find_first_not_of(" \t\r\n") == std::string::npos) {
81 continue;
82 }
83
84 // Skip comments
85 auto non_ws = line.find_first_not_of(" \t");
86 if (non_ws != std::string::npos && line[non_ws] == '#') {
87 continue;
88 }
89
90 // Calculate indentation level
91 int indent = 0;
92 for (char c : line) {
93 if (c == ' ') indent++;
94 else if (c == '\t') indent += 2;
95 else break;
96 }
97
98 // Trim the line
99 auto trimmed = trim(line);
100
101 // Check for list item
102 if (trimmed.starts_with("- ")) {
103 auto item = trim(trimmed.substr(2));
104 item = strip_quotes(item);
105 if (!current_list_path.empty()) {
106 lists_[current_list_path].push_back(item);
107 }
108 continue;
109 }
110
111 // Reset list context if not a list item
112 current_list_path.clear();
113
114 // Parse key: value
115 auto colon_pos = trimmed.find(':');
116 if (colon_pos == std::string::npos) continue;
117
118 std::string key = trim(trimmed.substr(0, colon_pos));
119 std::string value = (colon_pos + 1 < trimmed.size())
120 ? trim(trimmed.substr(colon_pos + 1))
121 : "";
122
123 // Pop path stack to appropriate level
124 while (!path_stack.empty() && path_stack.back().first >= indent) {
125 path_stack.pop_back();
126 }
127
128 // Build full path
129 std::string full_path;
130 for (const auto& [_, segment] : path_stack) {
131 full_path += segment + ".";
132 }
133 full_path += key;
134
135 if (value.empty()) {
136 // This is a section header
137 path_stack.emplace_back(indent, key);
138 current_list_path = full_path;
139 } else {
140 // This is a value
141 value = strip_quotes(value);
142 values_[full_path] = value;
143 }
144 }
145 }
146
147 [[nodiscard]] static auto trim(std::string_view str) -> std::string {
148 auto start = str.find_first_not_of(" \t\r\n");
149 if (start == std::string::npos) return "";
150 auto end = str.find_last_not_of(" \t\r\n");
151 return std::string(str.substr(start, end - start + 1));
152 }
153
154 [[nodiscard]] static auto strip_quotes(std::string_view str) -> std::string {
155 if (str.length() >= 2) {
156 if ((str.front() == '"' && str.back() == '"') ||
157 (str.front() == '\'' && str.back() == '\'')) {
158 return std::string(str.substr(1, str.length() - 2));
159 }
160 }
161 return std::string(str);
162 }

205 config.server.tls.ca_certificate = parser.get_string("server.tls.ca_certificate");
206 config.server.tls.require_client_cert =

208 config.server.tls.min_version = parser.get_string("server.tls.min_version", "1.2");

210 // Storage configuration

219
220 // Security configuration

225 config.security.allowed_ae_titles =

227 config.security.anonymization.auto_anonymize =
228 parser.get_bool("security.anonymization.auto_anonymize", false);
229
230 auto profile_str = parser.get_string("security.anonymization.profile", "basic");
231 config.security.anonymization.profile = parse_anonymization_profile(profile_str);
232
233 // REST API configuration
234 config.rest_api.enabled = parser.get_bool("rest_api.enabled", true);
235 config.rest_api.port = static_cast<std::uint16_t>(
236 parser.get_int("rest_api.port", config.rest_api.port));
237 config.rest_api.cors_enabled = parser.get_bool("rest_api.cors_enabled", true);
238
239 // Monitoring configuration
240 config.monitoring.health_check_interval = std::chrono::seconds(
241 parser.get_int("monitoring.health_check_interval_seconds", 30));
242 config.monitoring.metrics_enabled = parser.get_bool("monitoring.metrics_enabled", true);
243
244 // Logging configuration
245 config.logging.level = parser.get_string("logging.level", "info");
246 config.logging.audit_log_path = parser.get_string("logging.audit_log_path");
247
248 // Validate configuration
249 auto validation = validate(config);
250 if (!validation.is_ok()) {
251 return kcenon::common::Result<production_config>::err(
252 kcenon::common::error_info(validation.error().message));
253 }

255 return kcenon::common::Result<production_config>::ok(std::move(config));
256
257 } catch (const std::exception& e) {
258 return kcenon::common::Result<production_config>::err(
259 kcenon::common::error_info(std::string("Configuration parsing error: ") + e.what()));
260 }

300auto config_loader::validate(const production_config& config)
301 -> kcenon::common::VoidResult {
302 // Validate AE Title (max 16 characters, no leading/trailing spaces)
303 if (config.server.ae_title.empty()) {
304 return kcenon::common::VoidResult::err(
305 kcenon::common::error_info("AE Title cannot be empty"));
306 }
307 if (config.server.ae_title.length() > 16) {
308 return kcenon::common::VoidResult::err(
309 kcenon::common::error_info("AE Title exceeds maximum length of 16 characters"));
310 }
311
312 // Validate port
313 if (config.server.port == 0) {
314 return kcenon::common::VoidResult::err(
315 kcenon::common::error_info("Server port cannot be 0"));
316 }
317
318 // Validate REST API port
319 if (config.rest_api.enabled && config.rest_api.port == 0) {
320 return kcenon::common::VoidResult::err(
321 kcenon::common::error_info("REST API port cannot be 0"));
322 }
323
324 // Validate TLS configuration
325 if (config.server.tls.enabled) {
326 if (config.server.tls.certificate.empty()) {
327 return kcenon::common::VoidResult::err(
328 kcenon::common::error_info("TLS enabled but no certificate specified"));
329 }
330 if (config.server.tls.private_key.empty()) {
331 return kcenon::common::VoidResult::err(
332 kcenon::common::error_info("TLS enabled but no private key specified"));
333 }
334 }
335
336 // Validate naming scheme
337 const std::vector<std::string> valid_schemes = {
338 "uid_flat", "uid_hierarchical", "date_based"
339 };
340 auto scheme_it = std::find(valid_schemes.begin(), valid_schemes.end(),
341 config.storage.naming_scheme);
342 if (scheme_it == valid_schemes.end()) {
343 return kcenon::common::VoidResult::err(
344 kcenon::common::error_info("Invalid naming scheme: " + config.storage.naming_scheme));
345 }
346
347 // Validate duplicate policy
348 const std::vector<std::string> valid_policies = {"reject", "replace", "rename"};
349 auto policy_it = std::find(valid_policies.begin(), valid_policies.end(),
350 config.storage.duplicate_policy);
351 if (policy_it == valid_policies.end()) {
352 return kcenon::common::VoidResult::err(
353 kcenon::common::error_info("Invalid duplicate policy: " + config.storage.duplicate_policy));
354 }
355
356 return kcenon::common::VoidResult::ok(std::monostate{});
357}

378auto config_loader::parse_anonymization_profile(std::string_view value)
379 -> security::anonymization_profile {
380 auto v = trim(value);
381 std::transform(v.begin(), v.end(), v.begin(), ::tolower);
382
383 if (v == "basic" || v == "basic_profile") {
384 return security::anonymization_profile::basic;
385 }
386 if (v == "hipaa" || v == "hipaa_safe_harbor") {
387 return security::anonymization_profile::hipaa_safe_harbor;
388 }
389 if (v == "retain_longitudinal" || v == "retain_longitudinal_full_dates") {
390 return security::anonymization_profile::retain_longitudinal;
391 }
392 if (v == "retain_device" || v == "retain_device_identity" ||
393 v == "retain_patient" || v == "retain_patient_characteristics") {
394 return security::anonymization_profile::retain_patient_characteristics;
395 }
396 if (v == "gdpr") {
397 return security::anonymization_profile::gdpr_compliant;
398 }
399
400 // Default to basic
401 return security::anonymization_profile::basic;
402}
