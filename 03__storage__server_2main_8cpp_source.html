54std::string current_timestamp() {
55 auto now = std::chrono::system_clock::now();
56 auto time_t_now = std::chrono::system_clock::to_time_t(now);
57 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
58 now.time_since_epoch()) % 1000;
59
60 std::ostringstream oss;
61 oss << std::put_time(std::localtime(&time_t_now), "%H:%M:%S");
62 oss << '.' << std::setfill('0') << std::setw(3) << ms.count();
63 return oss.str();
64}

80bool update_index(stg::index_database& db,
81 const core::dicom_dataset& ds,
82 const std::filesystem::path& file_path) {
83 // Patient information
84 auto patient_result = db.upsert_patient(
85 ds.get_string(tags::patient_id),
86 ds.get_string(tags::patient_name),
87 ds.get_string(tags::patient_birth_date),
88 ds.get_string(tags::patient_sex)
89 );
90
91 if (patient_result.is_err()) {
92 std::cerr << " Database error (patient): "
93 << patient_result.error().message << "\n";
94 return false;
95 }
96 int64_t patient_pk = patient_result.value();
97
98 // Study information
99 auto study_result = db.upsert_study(
100 patient_pk,
101 ds.get_string(tags::study_instance_uid),
102 ds.get_string(tags::study_id),
103 ds.get_string(tags::study_date),
104 ds.get_string(tags::study_time),
105 ds.get_string(tags::accession_number),
106 ds.get_string(tags::referring_physician_name),
107 ds.get_string(tags::study_description)
108 );
109
110 if (study_result.is_err()) {
111 std::cerr << " Database error (study): "
112 << study_result.error().message << "\n";
113 return false;
114 }
115 int64_t study_pk = study_result.value();
116
117 // Series information
118 // Note: body_part_examined tag not defined in current core library,
119 // pass empty string as placeholder
120 auto series_result = db.upsert_series(
121 study_pk,
122 ds.get_string(tags::series_instance_uid),
123 ds.get_string(tags::modality),
124 ds.get_numeric<int>(tags::series_number),
125 ds.get_string(tags::series_description),
126 "", // body_part_examined - not available in current tag definitions
127 ds.get_string(tags::station_name)
128 );
129
130 if (series_result.is_err()) {
131 std::cerr << " Database error (series): "
132 << series_result.error().message << "\n";
133 return false;
134 }
135 int64_t series_pk = series_result.value();
136
137 // Instance information
138 auto file_size = std::filesystem::exists(file_path)
139 ? static_cast<int64_t>(std::filesystem::file_size(file_path))
140 : 0;
141
142 auto instance_result = db.upsert_instance(
143 series_pk,
144 ds.get_string(tags::sop_instance_uid),
145 ds.get_string(tags::sop_class_uid),
146 file_path.string(),
147 file_size,
148 ds.get_string(tags::transfer_syntax_uid),
149 ds.get_numeric<int>(tags::instance_number)
150 );
151
152 if (instance_result.is_err()) {
153 std::cerr << " Database error (instance): "
154 << instance_result.error().message << "\n";
155 return false;
156 }
157
158 return true;
159}
