34constexpr char kBase64Alphabet[] =
35 "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
36
37// Decoded value 0-63, or -1 on invalid. '=' is handled separately.
38int base64_value(char c) noexcept {
39 if (c >= 'A' && c <= 'Z') return c - 'A';
40 if (c >= 'a' && c <= 'z') return c - 'a' + 26;
41 if (c >= '0' && c <= '9') return c - '0' + 52;
42 if (c == '+') return 62;
43 if (c == '/') return 63;
44 return -1;
45}
46
47int hex_value(char c) noexcept {
48 if (c >= '0' && c <= '9') return c - '0';
49 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
50 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
51 return -1;
52}

56std::string base64_encode(const std::uint8_t* data, std::size_t size) {
57 std::string out;
58 out.reserve(((size + 2) / 3) * 4);
59
60 std::size_t i = 0;
61 for (; i + 3 <= size; i += 3) {
62 std::uint32_t v = (static_cast<std::uint32_t>(data[i]) << 16) |
63 (static_cast<std::uint32_t>(data[i + 1]) << 8) |
64 static_cast<std::uint32_t>(data[i + 2]);
65 out.push_back(kBase64Alphabet[(v >> 18) & 0x3F]);
66 out.push_back(kBase64Alphabet[(v >> 12) & 0x3F]);
67 out.push_back(kBase64Alphabet[(v >> 6) & 0x3F]);
68 out.push_back(kBase64Alphabet[v & 0x3F]);
69 }
70
71 const std::size_t rem = size - i;
72 if (rem == 1) {
73 std::uint32_t v = static_cast<std::uint32_t>(data[i]) << 16;
74 out.push_back(kBase64Alphabet[(v >> 18) & 0x3F]);
75 out.push_back(kBase64Alphabet[(v >> 12) & 0x3F]);
76 out.push_back('=');
77 out.push_back('=');
78 } else if (rem == 2) {
79 std::uint32_t v = (static_cast<std::uint32_t>(data[i]) << 16) |
80 (static_cast<std::uint32_t>(data[i + 1]) << 8);
81 out.push_back(kBase64Alphabet[(v >> 18) & 0x3F]);
82 out.push_back(kBase64Alphabet[(v >> 12) & 0x3F]);
83 out.push_back(kBase64Alphabet[(v >> 6) & 0x3F]);
84 out.push_back('=');
85 }
86 return out;
87}

89bool base64_decode(const std::string& in, std::vector<std::uint8_t>& out) {
90 if (in.size() % 4 != 0) return false;
91 out.clear();
92 out.reserve((in.size() / 4) * 3);
93
94 for (std::size_t i = 0; i < in.size(); i += 4) {
95 const char c0 = in[i];
96 const char c1 = in[i + 1];
97 const char c2 = in[i + 2];
98 const char c3 = in[i + 3];
99
100 const int v0 = base64_value(c0);
101 const int v1 = base64_value(c1);
102 if (v0 < 0 || v1 < 0) return false;
103
104 const std::uint32_t b0 = static_cast<std::uint32_t>(v0);
105 const std::uint32_t b1 = static_cast<std::uint32_t>(v1);
106
107 out.push_back(static_cast<std::uint8_t>((b0 << 2) | (b1 >> 4)));
108
109 if (c2 == '=') {
110 if (c3 != '=') return false;
111 if (i + 4 != in.size()) return false;
112 continue;
113 }
114 const int v2 = base64_value(c2);
115 if (v2 < 0) return false;
116 const std::uint32_t b2 = static_cast<std::uint32_t>(v2);
117
118 out.push_back(static_cast<std::uint8_t>(((b1 & 0xF) << 4) | (b2 >> 2)));
119
120 if (c3 == '=') {
121 if (i + 4 != in.size()) return false;
122 continue;
123 }
124 const int v3 = base64_value(c3);
125 if (v3 < 0) return false;
126 const std::uint32_t b3 = static_cast<std::uint32_t>(v3);
127 out.push_back(static_cast<std::uint8_t>(((b2 & 0x3) << 6) | b3));
128 }
129 return true;
130}
131
132bool hex_decode(const std::string& in,
133 std::uint8_t* out,
134 std::size_t out_size) {
135 if (in.size() != out_size * 2) return false;
136 for (std::size_t i = 0; i < out_size; ++i) {
137 const int hi = hex_value(in[2 * i]);
138 const int lo = hex_value(in[2 * i + 1]);
139 if (hi < 0 || lo < 0) return false;
140 out[i] = static_cast<std::uint8_t>((hi << 4) | lo);
141 }
142 return true;
143}

145bool constant_time_equal(const std::uint8_t* a,
146 const std::uint8_t* b,
147 std::size_t n) noexcept {
148 std::uint8_t diff = 0;
149 for (std::size_t i = 0; i < n; ++i) {
150 diff |= static_cast<std::uint8_t>(a[i] ^ b[i]);
151 }
152 return diff == 0;
153}

161void audit_cipher_key::wipe() noexcept {
162 // volatile write to discourage the compiler from optimising it out
163 volatile std::uint8_t* p = data_key.data();
164 for (std::size_t i = 0; i < data_key.size(); ++i) p[i] = 0;
165 volatile std::uint8_t* q = mac_key.data();
166 for (std::size_t i = 0; i < mac_key.size(); ++i) q[i] = 0;
167}

229VoidResult audit_key_manager::decode_key(const std::string& encoded,
230 std::uint8_t* out,
231 std::size_t out_size) {
232 if (encoded.empty() || out == nullptr || out_size == 0) {
233 return pacs_void_error(
234 kcenon::common::error::codes::common_errors::invalid_argument,
235 "empty key encoding or invalid buffer");
236 }
237
238 // Hex first: encoded length must be exactly 2 * out_size
239 if (encoded.size() == out_size * 2) {
240 if (detail::hex_decode(encoded, out, out_size)) {
241 return kcenon::common::ok();
242 }
243 // fall through — may still be base64 of the same length
244 }
245
246 std::vector<std::uint8_t> decoded;
247 if (!detail::base64_decode(encoded, decoded)) {
248 return pacs_void_error(
249 kcenon::common::error::codes::common_errors::invalid_argument,
250 "audit key is neither valid hex nor base64");
251 }
252 if (decoded.size() != out_size) {
253 return pacs_void_error(
254 kcenon::common::error::codes::common_errors::invalid_argument,
255 "audit key has wrong length after decoding");
256 }
257 std::memcpy(out, decoded.data(), out_size);
258 // wipe temporary copy
259 volatile std::uint8_t* p = decoded.data();
260 for (std::size_t i = 0; i < decoded.size(); ++i) p[i] = 0;
261 return kcenon::common::ok();
262}

271std::vector<std::string> split_csv(const std::string& s) {
272 std::vector<std::string> out;
273 std::string tok;
274 for (char c : s) {
275 if (c == ',' || c == ' ') {
276 if (!tok.empty()) {
277 out.push_back(tok);
278 tok.clear();
279 }
280 } else {
281 tok.push_back(c);
282 }
283 }
284 if (!tok.empty()) out.push_back(tok);
285 return out;
286}

290VoidResult audit_key_manager::load_from_env() {
291 const std::string active = getenv_str("PACS_AUDIT_LOG_ACTIVE_KEY_ID");
292 if (active.empty()) {
293 return pacs_void_error(
294 kcenon::common::error::codes::common_errors::invalid_argument,
295 "PACS_AUDIT_LOG_ACTIVE_KEY_ID is not set");
296 }
297
298 std::vector<std::string> ids;
299 const std::string list = getenv_str("PACS_AUDIT_LOG_KEYS");
300 if (list.empty()) {
301 ids.push_back(active);
302 } else {
303 ids = split_csv(list);
304 if (std::find(ids.begin(), ids.end(), active) == ids.end()) {
305 ids.push_back(active);
306 }
307 }
308
309 for (const auto& id : ids) {
310 if (id.empty() || id.find('|') != std::string::npos) {
311 return pacs_void_error(
312 kcenon::common::error::codes::common_errors::invalid_argument,
313 "invalid audit key id: '" + id + "'");
314 }
315 const std::string key_env = "PACS_AUDIT_LOG_KEY_" + id;
316 const std::string mac_env = "PACS_AUDIT_LOG_MAC_" + id;
317 const std::string key_val = getenv_str(key_env.c_str());
318 const std::string mac_val = getenv_str(mac_env.c_str());
319 if (key_val.empty() || mac_val.empty()) {
320 return pacs_void_error(
321 kcenon::common::error::codes::common_errors::invalid_argument,
322 "missing env var for audit key '" + id + "'");
323 }
324
325 audit_cipher_key k;
326 k.key_id = id;
327 auto r1 = decode_key(key_val, k.data_key.data(), k.data_key.size());
328 if (!r1.is_ok()) {
329 return pacs_void_error(
330 kcenon::common::error::codes::common_errors::invalid_argument,
331 "failed to decode " + key_env);
332 }
333 auto r2 = decode_key(mac_val, k.mac_key.data(), k.mac_key.size());
334 if (!r2.is_ok()) {
335 return pacs_void_error(
336 kcenon::common::error::codes::common_errors::invalid_argument,
337 "failed to decode " + mac_env);
338 }
339 register_key(std::move(k));
340 }
341
342 return set_active(active);
343}

362// RAII helper around EVP_CIPHER_CTX
363struct evp_cipher_ctx_guard {
364 EVP_CIPHER_CTX* ctx{nullptr};
365 evp_cipher_ctx_guard() : ctx(EVP_CIPHER_CTX_new()) {}
366 ~evp_cipher_ctx_guard() {
367 if (ctx) EVP_CIPHER_CTX_free(ctx);
368 }
369 evp_cipher_ctx_guard(const evp_cipher_ctx_guard&) = delete;
370 evp_cipher_ctx_guard& operator=(const evp_cipher_ctx_guard&) = delete;
371};

373bool hmac_sha256(const std::uint8_t* key,
374 std::size_t key_size,
375 const std::uint8_t* data,
376 std::size_t data_size,
377 std::uint8_t* out32) {
378 unsigned int out_len = 0;
379 const unsigned char* result = ::HMAC(
380 EVP_sha256(),
381 key, static_cast<int>(key_size),
382 data, data_size,
383 out32, &out_len);
384 return result != nullptr && out_len == 32;
385}

389Result<std::string> audit_log_cipher::encrypt_record(
390 const std::string& plaintext) const {
391 auto key_r = keys_.active();
392 if (!key_r.is_ok()) {
393 return pacs_error<std::string>(
394 kcenon::common::error::codes::common_errors::not_found,
395 "no active audit encryption key");
396 }
397 const audit_cipher_key& key = key_r.value();
398
399 if (key.key_id.empty() ||
400 key.key_id.find('|') != std::string::npos) {
401 return pacs_error<std::string>(
402 kcenon::common::error::codes::common_errors::invalid_argument,
403 "active audit key has invalid id");
404 }
405
406 // 1. Fresh random IV
407 std::array<std::uint8_t, audit_cipher_iv_bytes> iv{};
408 if (RAND_bytes(iv.data(), static_cast<int>(iv.size())) != 1) {
409 return pacs_error<std::string>(
410 kcenon::common::error::codes::common_errors::internal_error,
411 "RAND_bytes failed for audit log IV");
412 }
413
414 // 2. AES-256-GCM encrypt
415 evp_cipher_ctx_guard guard;
416 if (!guard.ctx) {
417 return pacs_error<std::string>(
418 kcenon::common::error::codes::common_errors::internal_error,
419 "EVP_CIPHER_CTX_new failed");
420 }
421 if (EVP_EncryptInit_ex(guard.ctx, EVP_aes_256_gcm(), nullptr,
422 key.data_key.data(), iv.data()) != 1) {
423 return pacs_error<std::string>(
424 kcenon::common::error::codes::common_errors::internal_error,
425 "EVP_EncryptInit_ex failed");
426 }
427
428 std::vector<std::uint8_t> ct(plaintext.size() + 16); // GCM no padding
429 int out_len = 0;
430 if (!plaintext.empty()) {
431 if (EVP_EncryptUpdate(
432 guard.ctx, ct.data(), &out_len,
433 reinterpret_cast<const unsigned char*>(plaintext.data()),
434 static_cast<int>(plaintext.size())) != 1) {
435 return pacs_error<std::string>(
436 kcenon::common::error::codes::common_errors::internal_error,
437 "EVP_EncryptUpdate failed");
438 }
439 }
440 int final_len = 0;
441 if (EVP_EncryptFinal_ex(guard.ctx, ct.data() + out_len, &final_len) != 1) {
442 return pacs_error<std::string>(
443 kcenon::common::error::codes::common_errors::internal_error,
444 "EVP_EncryptFinal_ex failed");
445 }
446 ct.resize(static_cast<std::size_t>(out_len + final_len));
447
448 std::array<std::uint8_t, audit_cipher_tag_bytes> tag{};
449 if (EVP_CIPHER_CTX_ctrl(guard.ctx, EVP_CTRL_GCM_GET_TAG,
450 static_cast<int>(tag.size()),
451 tag.data()) != 1) {
452 return pacs_error<std::string>(
453 kcenon::common::error::codes::common_errors::internal_error,
454 "EVP_CTRL_GCM_GET_TAG failed");
455 }
456
457 // 3. Serialise prefix and inner fields
458 const std::string iv_b64 = detail::base64_encode(iv.data(), iv.size());
459 const std::string ct_b64 = detail::base64_encode(ct.data(), ct.size());
460 const std::string tag_b64 = detail::base64_encode(tag.data(), tag.size());
461
462 std::string prefix;
463 prefix.reserve(64 + key.key_id.size() + iv_b64.size() +
464 ct_b64.size() + tag_b64.size());
465 prefix.append(audit_cipher_format_tag);
466 prefix.push_back('|');
467 prefix.append(audit_cipher_format_version);
468 prefix.push_back('|');
469 prefix.append(key.key_id);
470 prefix.push_back('|');
471 prefix.append(iv_b64);
472 prefix.push_back('|');
473 prefix.append(ct_b64);
474 prefix.push_back('|');
475 prefix.append(tag_b64);
476
477 // 4. HMAC-SHA256 over prefix, using the independent MAC key
478 std::array<std::uint8_t, 32> mac{};
479 if (!hmac_sha256(key.mac_key.data(), key.mac_key.size(),
480 reinterpret_cast<const std::uint8_t*>(prefix.data()),
481 prefix.size(),
482 mac.data())) {
483 return pacs_error<std::string>(
484 kcenon::common::error::codes::common_errors::internal_error,
485 "HMAC-SHA256 failed");
486 }
487 const std::string mac_b64 = detail::base64_encode(mac.data(), mac.size());
488
489 prefix.push_back('|');
490 prefix.append(mac_b64);
491 return kcenon::common::Result<std::string>(std::move(prefix));
492}

496struct parsed_record {
497 std::string key_id;
498 std::vector<std::uint8_t> iv;
499 std::vector<std::uint8_t> ciphertext;
500 std::vector<std::uint8_t> tag;
501 std::vector<std::uint8_t> mac;
502 std::string mac_prefix; // the exact bytes that were HMAC'd
503};
504
505// Parse "PACSAUDIT|v1|<id>|<iv>|<ct>|<tag>|<hmac>", trailing '\n' tolerated.
506bool parse_record(std::string line, parsed_record& out) {
507 while (!line.empty() &&
508 (line.back() == '\n' || line.back() == '\r')) {
509 line.pop_back();
510 }
511
512 // Split on '|' into exactly 7 fields.
513 std::array<std::string, 7> fields;
514 std::size_t idx = 0;
515 std::string tok;
516 for (char c : line) {
517 if (c == '|') {
518 if (idx >= fields.size()) return false;
519 fields[idx++] = std::move(tok);
520 tok.clear();
521 } else {
522 tok.push_back(c);
523 }
524 }
525 if (idx >= fields.size()) return false;
526 fields[idx++] = std::move(tok);
527 if (idx != fields.size()) return false;
528
529 if (fields[0] != audit_cipher_format_tag) return false;
530 if (fields[1] != audit_cipher_format_version) return false;
531 if (fields[2].empty()) return false;
532
533 if (!detail::base64_decode(fields[3], out.iv)) return false;
534 if (!detail::base64_decode(fields[4], out.ciphertext)) return false;
535 if (!detail::base64_decode(fields[5], out.tag)) return false;
536 if (!detail::base64_decode(fields[6], out.mac)) return false;
537
538 if (out.iv.size() != audit_cipher_iv_bytes) return false;
539 if (out.tag.size() != audit_cipher_tag_bytes) return false;
540 if (out.mac.size() != 32) return false;
541
542 out.key_id = fields[2];
543 // Reconstruct the MAC-covered prefix (everything before the final '|')
544 out.mac_prefix.reserve(line.size());
545 out.mac_prefix.append(fields[0]);
546 out.mac_prefix.push_back('|');
547 out.mac_prefix.append(fields[1]);
548 out.mac_prefix.push_back('|');
549 out.mac_prefix.append(fields[2]);
550 out.mac_prefix.push_back('|');
551 out.mac_prefix.append(fields[3]);
552 out.mac_prefix.push_back('|');
553 out.mac_prefix.append(fields[4]);
554 out.mac_prefix.push_back('|');
555 out.mac_prefix.append(fields[5]);
556 return true;
557}

561Result<std::string> audit_log_cipher::decrypt_record(
562 const std::string& encoded) const {
563 parsed_record rec;
564 if (!parse_record(encoded, rec)) {
565 return pacs_error<std::string>(
566 kcenon::common::error::codes::common_errors::invalid_argument,
567 "malformed encrypted audit record");
568 }
569
570 const audit_cipher_key* key = keys_.find(rec.key_id);
571 if (key == nullptr) {
572 return pacs_error<std::string>(
573 kcenon::common::error::codes::common_errors::not_found,
574 "audit record key id not registered: " + rec.key_id);
575 }
576
577 // 1. Verify HMAC first (fail fast before touching GCM)
578 std::array<std::uint8_t, 32> expected_mac{};
579 if (!hmac_sha256(key->mac_key.data(), key->mac_key.size(),
580 reinterpret_cast<const std::uint8_t*>(
581 rec.mac_prefix.data()),
582 rec.mac_prefix.size(),
583 expected_mac.data())) {
584 return pacs_error<std::string>(
585 kcenon::common::error::codes::common_errors::internal_error,
586 "HMAC-SHA256 failed during verify");
587 }
588 if (!detail::constant_time_equal(expected_mac.data(),
589 rec.mac.data(),
590 expected_mac.size())) {
591 return pacs_error<std::string>(
592 kcenon::common::error::codes::common_errors::invalid_argument,
593 "audit record HMAC verification failed");
594 }
595
596 // 2. AES-256-GCM decrypt
597 evp_cipher_ctx_guard guard;
598 if (!guard.ctx) {
599 return pacs_error<std::string>(
600 kcenon::common::error::codes::common_errors::internal_error,
601 "EVP_CIPHER_CTX_new failed");
602 }
603 if (EVP_DecryptInit_ex(guard.ctx, EVP_aes_256_gcm(), nullptr,
604 key->data_key.data(), rec.iv.data()) != 1) {
605 return pacs_error<std::string>(
606 kcenon::common::error::codes::common_errors::internal_error,
607 "EVP_DecryptInit_ex failed");
608 }
609
610 std::vector<std::uint8_t> pt(rec.ciphertext.size() + 16);
611 int out_len = 0;
612 if (!rec.ciphertext.empty()) {
613 if (EVP_DecryptUpdate(guard.ctx, pt.data(), &out_len,
614 rec.ciphertext.data(),
615 static_cast<int>(rec.ciphertext.size())) != 1) {
616 return pacs_error<std::string>(
617 kcenon::common::error::codes::common_errors::invalid_argument,
618 "EVP_DecryptUpdate failed");
619 }
620 }
621
622 if (EVP_CIPHER_CTX_ctrl(guard.ctx, EVP_CTRL_GCM_SET_TAG,
623 static_cast<int>(rec.tag.size()),
624 rec.tag.data()) != 1) {
625 return pacs_error<std::string>(
626 kcenon::common::error::codes::common_errors::internal_error,
627 "EVP_CTRL_GCM_SET_TAG failed");
628 }
629

637
638 return kcenon::common::Result<std::string>(
639 std::string(reinterpret_cast<const char*>(pt.data()), pt.size()));
640}
