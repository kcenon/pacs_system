139void mini_pacs::initialize_services() {
140 // Verification SCP (C-ECHO)
141 verification_scp_ = std::make_shared<services::verification_scp>();
142
143 // Storage SCP (C-STORE)
144 services::storage_scp_config scp_config;
145 storage_scp_ = std::make_shared<services::storage_scp>(scp_config);
146 storage_scp_->set_handler(
147 [this](const auto& ds, const auto& ae, const auto& sop_class,
148 const auto& sop_instance) {
149 return handle_store(ds, ae, sop_class, sop_instance);
150 });
151
152 // Query SCP (C-FIND)
153 query_scp_ = std::make_shared<services::query_scp>();
154 query_scp_->set_handler(
155 [this](auto level, const auto& keys, const auto& ae) {
156 return handle_query(level, keys, ae);
157 });
158
159 // Retrieve SCP (C-MOVE/C-GET)
160 retrieve_scp_ = std::make_shared<services::retrieve_scp>();
161 retrieve_scp_->set_retrieve_handler(
162 [this](const auto& keys) { return handle_retrieve(keys); });
163 retrieve_scp_->set_destination_resolver(
164 [this](const auto& ae) { return resolve_destination(ae); });
165
166 // Worklist SCP (MWL C-FIND)
167 if (config_.enable_worklist) {
168 worklist_scp_ = std::make_shared<services::worklist_scp>();
169 worklist_scp_->set_handler(
170 [this](const auto& keys, const auto& ae) {
171 return handle_worklist_query(keys, ae);
172 });
173 }
174
175 // MPPS SCP (N-CREATE/N-SET)
176 if (config_.enable_mpps) {
177 mpps_scp_ = std::make_shared<services::mpps_scp>();
178 mpps_scp_->set_create_handler(
179 [this](const auto& inst) { return handle_mpps_create(inst); });
180 mpps_scp_->set_set_handler(
181 [this](const auto& uid, const auto& mods, auto status) {
182 return handle_mpps_set(uid, mods, status);
183 });
184 }

358std::vector<core::dicom_dataset> mini_pacs::handle_query(
359 services::query_level level,
360 const core::dicom_dataset& query_keys,
361 const std::string& /*calling_ae*/) {
362
363 stats_.c_find_count++;
364
365 std::vector<core::dicom_dataset> results;
366
367 switch (level) {
368 case services::query_level::patient: {
369 storage::patient_query pq;
370 auto pid = query_keys.get_string(tags::patient_id);
371 auto pname = query_keys.get_string(tags::patient_name);
372 if (!pid.empty()) pq.patient_id = pid;
373 if (!pname.empty()) pq.patient_name = pname;
374
375 auto patients = index_db_->search_patients(pq);
376
377 if (patients.is_ok()) {
378 for (const auto& p : patients.value()) {
379 core::dicom_dataset ds;
380 ds.set_string(tags::patient_id, encoding::vr_type::LO, p.patient_id);
381 ds.set_string(tags::patient_name, encoding::vr_type::PN, p.patient_name);
382 ds.set_string(tags::patient_birth_date, encoding::vr_type::DA, p.birth_date);
383 ds.set_string(tags::patient_sex, encoding::vr_type::CS, p.sex);
384 results.push_back(std::move(ds));
385 }
386 }
387 break;
388 }
389
390 case services::query_level::study: {
391 storage::study_query sq;
392 auto pid = query_keys.get_string(tags::patient_id);
393 auto study_uid = query_keys.get_string(tags::study_instance_uid);
394 auto study_date = query_keys.get_string(tags::study_date);
395 auto accession = query_keys.get_string(tags::accession_number);
396 if (!pid.empty()) sq.patient_id = pid;
397 if (!study_uid.empty()) sq.study_uid = study_uid;
398 if (!study_date.empty()) sq.study_date = study_date;
399 if (!accession.empty()) sq.accession_number = accession;
400
401 auto studies = index_db_->search_studies(sq);
402
403 if (studies.is_ok()) {
404 for (const auto& s : studies.value()) {
405 core::dicom_dataset ds;
406 ds.set_string(tags::study_instance_uid, encoding::vr_type::UI, s.study_uid);
407 ds.set_string(tags::study_id, encoding::vr_type::SH, s.study_id);
408 ds.set_string(tags::study_date, encoding::vr_type::DA, s.study_date);
409 ds.set_string(tags::study_time, encoding::vr_type::TM, s.study_time);
410 ds.set_string(tags::accession_number, encoding::vr_type::SH, s.accession_number);
411 ds.set_string(tags::study_description, encoding::vr_type::LO, s.study_description);
412 ds.set_string(tags::referring_physician_name, encoding::vr_type::PN, s.referring_physician);
413 results.push_back(std::move(ds));
414 }
415 }
416 break;
417 }
418
419 case services::query_level::series: {
420 storage::series_query sq;
421 auto study_uid = query_keys.get_string(tags::study_instance_uid);
422 auto series_uid = query_keys.get_string(tags::series_instance_uid);
423 auto modality = query_keys.get_string(tags::modality);
424 if (!study_uid.empty()) sq.study_uid = study_uid;
425 if (!series_uid.empty()) sq.series_uid = series_uid;
426 if (!modality.empty()) sq.modality = modality;
427
428 auto series = index_db_->search_series(sq);
429
430 if (series.is_ok()) {
431 for (const auto& ser : series.value()) {
432 core::dicom_dataset ds;
433 ds.set_string(tags::series_instance_uid, encoding::vr_type::UI, ser.series_uid);
434 ds.set_string(tags::modality, encoding::vr_type::CS, ser.modality);
435 if (ser.series_number.has_value()) {
436 ds.set_numeric(tags::series_number, encoding::vr_type::IS, ser.series_number.value());
437 }
438 ds.set_string(tags::series_description, encoding::vr_type::LO, ser.series_description);
439 ds.set_string(tags::station_name, encoding::vr_type::SH, ser.station_name);
440 results.push_back(std::move(ds));
441 }
442 }
443 break;
444 }
445
446 case services::query_level::image: {
447 storage::instance_query iq;
448 auto series_uid = query_keys.get_string(tags::series_instance_uid);
449 auto sop_uid = query_keys.get_string(tags::sop_instance_uid);
450 if (!series_uid.empty()) iq.series_uid = series_uid;
451 if (!sop_uid.empty()) iq.sop_uid = sop_uid;
452
453 auto instances = index_db_->search_instances(iq);
454
455 if (instances.is_ok()) {
456 for (const auto& inst : instances.value()) {
457 core::dicom_dataset ds;
458 ds.set_string(tags::sop_instance_uid, encoding::vr_type::UI, inst.sop_uid);
459 ds.set_string(tags::sop_class_uid, encoding::vr_type::UI, inst.sop_class_uid);
460 if (inst.instance_number.has_value()) {
461 ds.set_numeric(tags::instance_number, encoding::vr_type::IS, inst.instance_number.value());
462 }
463 results.push_back(std::move(ds));
464 }
465 }
466 break;
467 }
468 }
469
470 if (config_.verbose_logging) {
471 std::cout << "[C-FIND] Level: " << to_string(level)
472 << " Results: " << results.size() << "\n";
473 }
474
475 return results;
476}

482std::vector<core::dicom_file> mini_pacs::handle_retrieve(
483 const core::dicom_dataset& query_keys) {
484
485 std::vector<core::dicom_file> files;
486
487 // Determine query level from keys
488 auto study_uid = query_keys.get_string(tags::study_instance_uid);
489 auto series_uid = query_keys.get_string(tags::series_instance_uid);
490 auto sop_uid = query_keys.get_string(tags::sop_instance_uid);
491
492 std::vector<storage::instance_record> instances;
493
494 if (!sop_uid.empty()) {
495 // Instance level retrieve
496 storage::instance_query iq;
497 iq.sop_uid = sop_uid;
498 auto result = index_db_->search_instances(iq);
499 if (result.is_ok()) {
500 instances = result.value();
501 }
502 } else if (!series_uid.empty()) {
503 // Series level retrieve
504 storage::instance_query iq;
505 iq.series_uid = series_uid;
506 auto result = index_db_->search_instances(iq);
507 if (result.is_ok()) {
508 instances = result.value();
509 }
510 } else if (!study_uid.empty()) {
511 // Study level retrieve - get all series first
512 storage::series_query sq;
513 sq.study_uid = study_uid;
514 auto series_result = index_db_->search_series(sq);
515 if (series_result.is_ok()) {
516 for (const auto& ser : series_result.value()) {
517 storage::instance_query iq;
518 iq.series_uid = ser.series_uid;
519 auto inst_result = index_db_->search_instances(iq);
520 if (inst_result.is_ok()) {
521 for (const auto& inst : inst_result.value()) {
522 instances.push_back(inst);
523 }
524 }
525 }
526 }
527 }
528
529 // Load files from disk
530 for (const auto& inst : instances) {
531 auto file_result = core::dicom_file::open(inst.file_path);
532 if (file_result.is_ok()) {
533 files.push_back(std::move(file_result.value()));
534 }
535 }
536
537 stats_.c_move_count++;
538
539 if (config_.verbose_logging) {
540 std::cout << "[C-MOVE/C-GET] Files: " << files.size() << "\n";
541 }
542
543 return files;
544}

565 stats_.mwl_count++;

567 std::vector<core::dicom_dataset> results;

569 // Extract filter criteria

573
574 std::lock_guard<std::mutex> lock(worklist_mutex_);

579 continue;
580 }
581 if (!filter_date.empty() && item.scheduled_date != filter_date) {
582 continue;
583 }
584 if (!filter_patient_id.empty() && item.patient_id != filter_patient_id) {
585 continue;
586 }
587
588 results.push_back(create_worklist_response(item));
589 }
590
591 if (config_.verbose_logging) {
592 std::cout << "[MWL] Results: " << results.size() << "\n";
593 }
