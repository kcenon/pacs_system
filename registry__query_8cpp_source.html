29// ITI-18 RegistryStoredQuery SOAP action URI. The default
30// http_options::soap_action is the ITI-41 action, which would make the
31// registry reject the POST on the first line of WS-Addressing routing.
32// Registry Query overwrites it on a copy of the caller's options so the
33// caller's struct is preserved across calls.
34constexpr const char* kIti18SoapAction =
35 "urn:ihe:iti:2007:RegistryStoredQuery";
36
37// Content-Type of a plain SOAP 1.2 request. ITI-18 carries no attachments
38// in either direction, so the wire is a single application/soap+xml part.
39// Composed at compile time from kIti18SoapAction so the action URI is not
40// duplicated (review m5).
41constexpr const char* kSoapContentType =
42 "application/soap+xml; charset=UTF-8; "
43 "action=\"urn:ihe:iti:2007:RegistryStoredQuery\"";
44
45// Public-API input-validation predicate: rejects ebRIM StoredQuery
46// slot-literal metacharacters. Kept at the public boundary so callers
47// see the invalid-input error without a transport or envelope call,
48// and so the rejection is visible to auditors tracing the public
49// surface (review M1). The envelope builder keeps its own copy of this
50// check as defense in depth.
51bool contains_slot_literal_metachar(std::string_view v) {
52 for (char c : v) {
53 if (c == '\'' || c == '(' || c == ')' || c == ',') return true;
54 if (static_cast<unsigned char>(c) < 0x20) return true;
55 }
56 return false;
57}
58
59// Map the generic envelope signer/verifier failure codes to the
60// query-specific namespace so callers can branch on
61// registry_query_signature_* without pulling the Consumer vocabulary.
62int map_signature_error(int raw) {
63 if (raw ==
64 static_cast<int>(error_code::consumer_signature_verification_failed)) {
65 return static_cast<int>(
66 error_code::registry_query_signature_verification_failed);
67 }
68 if (raw == static_cast<int>(error_code::consumer_signature_missing)) {
69 return static_cast<int>(error_code::registry_query_signature_missing);
70 }
71 return raw;
72}

86 kcenon::common::Result<registry_query_result> find_documents(
87 const std::string& patient_id,
88 const registry_query_options& options) {
89 auto do_audit =
90 [&](const kcenon::common::Result<registry_query_result>& r) {
91 audit_find(patient_id, r);
92 return r;
93 };
94
95 if (patient_id.empty()) {
96 return do_audit(kcenon::common::make_error<registry_query_result>(
97 static_cast<int>(
98 error_code::registry_query_missing_patient_id),
99 "find_documents: patient_id is empty",
100 std::string(error_source)));
101 }
102 if (contains_slot_literal_metachar(patient_id)) {
103 return do_audit(kcenon::common::make_error<registry_query_result>(
104 static_cast<int>(
105 error_code::registry_query_invalid_patient_id),
106 "find_documents: patient_id contains a character that is "
107 "not legal inside an ebRIM StoredQuery slot literal "
108 "('(),\\' or control character)",
109 std::string(error_source)));
110 }
111 for (const auto& s : options.status_values) {
112 if (contains_slot_literal_metachar(s)) {
113 return do_audit(
114 kcenon::common::make_error<registry_query_result>(
115 static_cast<int>(
116 error_code::registry_query_invalid_patient_id),
117 "find_documents: options.status_values contains a "
118 "slot-literal metacharacter",
119 std::string(error_source)));
120 }
121 }
122 if (contains_slot_literal_metachar(options.creation_time_from) ||
123 contains_slot_literal_metachar(options.creation_time_to)) {
124 return do_audit(kcenon::common::make_error<registry_query_result>(
125 static_cast<int>(
126 error_code::registry_query_invalid_patient_id),
127 "find_documents: options.creation_time_* contains a "
128 "slot-literal metacharacter",
129 std::string(error_source)));
130 }
131
132 registry_query_request req;
133 req.patient_id = patient_id;
134 req.options = options;
135
136 return do_audit(
137 execute(req, detail::stored_query_kind::find_documents));
138 }

140 kcenon::common::Result<registry_query_result> get_documents(
141 const std::vector<std::string>& uuids) {
142 auto do_audit =
143 [&](const kcenon::common::Result<registry_query_result>& r) {
144 audit_get(uuids, r);
145 return r;
146 };
147
148 if (uuids.empty()) {
149 return do_audit(kcenon::common::make_error<registry_query_result>(
150 static_cast<int>(
151 error_code::registry_query_empty_uuid_list),
152 "get_documents: uuids is empty",
153 std::string(error_source)));
154 }
155 for (const auto& u : uuids) {
156 if (u.empty()) {
157 return do_audit(
158 kcenon::common::make_error<registry_query_result>(
159 static_cast<int>(
160 error_code::registry_query_missing_document_uuid),
161 "get_documents: uuids contains an empty element",
162 std::string(error_source)));
163 }
164 if (contains_slot_literal_metachar(u)) {
165 return do_audit(
166 kcenon::common::make_error<registry_query_result>(
167 static_cast<int>(
168 error_code::registry_query_missing_document_uuid),
169 "get_documents: uuids contains an element with a "
170 "slot-literal metacharacter",
171 std::string(error_source)));
172 }
173 }
174
175 registry_query_request req;
176 req.document_uuids = uuids;
177
178 return do_audit(
179 execute(req, detail::stored_query_kind::get_documents));
180 }

182private:
183 kcenon::common::Result<registry_query_result> execute(
184 const registry_query_request& req,
185 detail::stored_query_kind kind) {
186 auto env_result = detail::build_iti18_envelope(req, kind);
187 if (env_result.is_err()) {
188 return kcenon::common::make_error<registry_query_result>(
189 env_result.error());
190 }
191 auto env = env_result.value();
192
193 auto sign_result = detail::sign_envelope(
194 env, signing_.certificate_pem, signing_.private_key_pem,
195 signing_.private_key_password);
196 if (sign_result.is_err()) {
197 return kcenon::common::make_error<registry_query_result>(
198 sign_result.error());
199 }
200
201 // Per-call copy: the ITI-18 action is specific to this transaction
202 // and must not mutate the shared opts_ member, so other transaction
203 // types sharing the same http_options template do not pick up a
204 // stale action from a previous call.
205 http_options call_opts = opts_;
206 call_opts.soap_action = kIti18SoapAction;
207
208 auto http_result =
209 detail::http_post(call_opts, kSoapContentType, env.xml);
210 if (http_result.is_err()) {
211 return kcenon::common::make_error<registry_query_result>(
212 http_result.error());
213 }
214 auto response = http_result.value();
215
216 // Verify the signature on the response envelope before trusting
217 // its status attribute or any entry it carries. An unsigned
218 // "Failure" response could otherwise be used to deny service or
219 // mask real successful queries; this check short-circuits before
220 // parse_query_response sees the body.
221 auto verify = detail::verify_envelope_integrity(response.body);
222 if (verify.is_err()) {
223 auto err = verify.error();
224 err.code = map_signature_error(err.code);
225 return kcenon::common::make_error<registry_query_result>(err);
226 }
227
228 return detail::parse_query_response(response.body);
229 }

231 void audit_find(const std::string& patient_id,
232 const kcenon::common::Result<registry_query_result>& r) {
233 if (!audit_sink_) return;
234 kcenon::pacs::security::xds_iti18_event e;
235 e.registry_endpoint = opts_.endpoint;
236 e.query_kind =
237 kcenon::pacs::security::xds_iti18_event::kind::find_documents;
238 e.patient_id = patient_id;
239 if (r.is_ok()) {
240 e.outcome = kcenon::pacs::security::atna_event_outcome::success;
241 } else {
242 e.outcome =
243 kcenon::pacs::security::atna_event_outcome::serious_failure;
244 e.failure_description = r.error().message;
245 }
246 audit_sink_->on_iti18_query(e);
247 }
