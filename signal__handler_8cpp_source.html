57void signal_handler::request_shutdown() {
58 bool expected = false;
59 if (shutdown_requested_.compare_exchange_strong(expected, true,
60 std::memory_order_release, std::memory_order_relaxed)) {
61 // Invoke callback only once
62 if (callback_ && !callback_invoked_.exchange(true, std::memory_order_acq_rel)) {
63 callback_();
