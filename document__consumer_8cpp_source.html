29// ITI-43 Retrieve Document Set-b SOAP action URI. The default
30// http_options::soap_action is the ITI-41 action, which would make the
31// registry reject the POST on the first line of WS-Addressing routing. The
32// Consumer overwrites it on a copy of the caller's options so the caller's
33// struct is preserved.
34constexpr const char* kIti43SoapAction = "urn:ihe:iti:2007:RetrieveDocumentSet";
35
36// Content-Type of a plain SOAP 1.2 request with no MTOM attachments. ITI-43
37// requests carry only the envelope; MTOM only appears on the response.
38constexpr const char* kSoapContentType =
39 "application/soap+xml; charset=UTF-8; "
40 "action=\"urn:ihe:iti:2007:RetrieveDocumentSet\"";

54 kcenon::common::Result<document_response> retrieve(
55 const std::string& document_unique_id,
56 const std::string& repository_unique_id) {
57 auto do_audit =
58 [&](const kcenon::common::Result<document_response>& r) {
59 audit(document_unique_id, repository_unique_id, r);
60 return r;
61 };
62
63 if (document_unique_id.empty()) {
64 return do_audit(kcenon::common::make_error<document_response>(
65 static_cast<int>(
66 error_code::consumer_retrieve_missing_document_id),
67 "retrieve: document_unique_id is empty",
68 std::string(error_source)));
69 }
70 if (repository_unique_id.empty()) {
71 return do_audit(kcenon::common::make_error<document_response>(
72 static_cast<int>(
73 error_code::consumer_retrieve_missing_repository_id),
74 "retrieve: repository_unique_id is empty",
75 std::string(error_source)));
76 }
77
78 retrieve_request req;
79 req.document_unique_id = document_unique_id;
80 req.repository_unique_id = repository_unique_id;
81
82 auto env_result = detail::build_iti43_envelope(req);
83 if (env_result.is_err()) {
84 return do_audit(kcenon::common::make_error<document_response>(
85 env_result.error()));
86 }
87 auto env = env_result.value();
88
89 auto sign_result = detail::sign_envelope(
90 env, signing_.certificate_pem, signing_.private_key_pem,
91 signing_.private_key_password);
92 if (sign_result.is_err()) {
93 return do_audit(kcenon::common::make_error<document_response>(
94 sign_result.error()));
95 }
96
97 // Per-call copy: the ITI-43 SOAP action is specific to this
98 // transaction and must not mutate the shared opts_ member, so
99 // other transaction types - if this Consumer later grows more
100 // retrieve variants - cannot pick up a stale action from a
101 // previous call. This also preserves whatever the caller set.
102 http_options call_opts = opts_;
103 call_opts.soap_action = kIti43SoapAction;
104
105 auto http_result =
106 detail::http_post(call_opts, kSoapContentType, env.xml);
107 if (http_result.is_err()) {
108 return do_audit(kcenon::common::make_error<document_response>(
109 http_result.error()));
110 }
111 auto response = http_result.value();
112
113 auto parsed =
114 detail::parse_mtom_response(response.content_type, response.body);
115 if (parsed.is_err()) {
116 return do_audit(kcenon::common::make_error<document_response>(
117 parsed.error()));
118 }
119 auto mtom = parsed.value();
120
121 auto verify = detail::verify_envelope_integrity(mtom.root_xml);
122 if (verify.is_err()) {
123 return do_audit(kcenon::common::make_error<document_response>(
124 verify.error()));
125 }
126
127 return do_audit(detail::parse_retrieve_response(
128 mtom.root_xml, mtom.parts, req));
129 }
130
131private:
132 void audit(const std::string& document_unique_id,
133 const std::string& repository_unique_id,
134 const kcenon::common::Result<document_response>& r) {
135 if (!audit_sink_) return;
136 kcenon::pacs::security::xds_iti43_event e;
137 e.source_endpoint = opts_.endpoint;
138 e.document_unique_id = document_unique_id;
139 e.repository_unique_id = repository_unique_id;
140 if (r.is_ok()) {
141 e.outcome = kcenon::pacs::security::atna_event_outcome::success;
142 } else {
143 e.outcome =
144 kcenon::pacs::security::atna_event_outcome::serious_failure;
145 e.failure_description = r.error().message;
146 }
147 audit_sink_->on_iti43_retrieve(e);
148 }
149
150 http_options opts_;
151 signing_material signing_;
152 std::shared_ptr<kcenon::pacs::security::xds_audit_sink> audit_sink_;
153};
