41auto supports_color() noexcept -> bool {
42 // Check if stdout is a terminal
43 if (!isatty(fileno(stdout))) {
44 return false;
45 }
46
47#ifdef _WIN32
48 // Enable virtual terminal processing on Windows 10+
49 HANDLE hOut = GetStdHandle(STD_OUTPUT_HANDLE);
50 if (hOut == INVALID_HANDLE_VALUE) {
51 return false;
52 }
53
54 DWORD dwMode = 0;
55 if (!GetConsoleMode(hOut, &dwMode)) {
56 return false;
57 }
58
59 dwMode |= ENABLE_VIRTUAL_TERMINAL_PROCESSING;
60 if (!SetConsoleMode(hOut, dwMode)) {
61 // Fallback: check if TERM is set (e.g., running in mintty)
62 const char* term = std::getenv("TERM");
63 return term != nullptr;
64 }
65 return true;
66#else
67 // Check common environment variables
68 const char* term = std::getenv("TERM");
69 if (term == nullptr) {
70 return false;
71 }
72
73 // Check for dumb terminal
74 if (std::string_view{term} == "dumb") {
75 return false;
76 }
77
78 // Check for NO_COLOR environment variable
79 if (std::getenv("NO_COLOR") != nullptr) {
80 return false;
81 }
82
83 return true;
84#endif
85}

228void print_box(const std::vector<std::string>& lines) {
229 if (lines.empty()) {
230 return;
231 }
232
233 // Find maximum line length
234 size_t max_len = 0;
235 for (const auto& line : lines) {
236 max_len = std::max(max_len, line.length());
237 }
238 max_len = std::min(max_len, static_cast<size_t>(kDefaultWidth - 4));
239
240 std::cout << color(colors::dim);
241 std::cout << "+" << std::string(max_len + 2, '-') << "+\n";
242 for (const auto& line : lines) {
243 std::cout << "| " << std::left << std::setw(static_cast<int>(max_len))
244 << truncate(line, max_len) << " |\n";
245 }
246 std::cout << "+" << std::string(max_len + 2, '-') << "+\n";
247 std::cout << color(colors::reset);
248}

310auto format_bytes(uint64_t bytes) -> std::string {
311 constexpr uint64_t KB = 1024;
312 constexpr uint64_t MB = KB * 1024;
313 constexpr uint64_t GB = MB * 1024;
314
315 std::ostringstream oss;
316 oss << std::fixed << std::setprecision(1);
317
318 if (bytes >= GB) {
319 oss << static_cast<double>(bytes) / static_cast<double>(GB) << " GB";
320 } else if (bytes >= MB) {
321 oss << static_cast<double>(bytes) / static_cast<double>(MB) << " MB";
322 } else if (bytes >= KB) {
323 oss << static_cast<double>(bytes) / static_cast<double>(KB) << " KB";
324 } else {
325 oss << bytes << " B";
326 }
327
328 return oss.str();
329}

331auto format_duration(uint64_t milliseconds) -> std::string {
332 std::ostringstream oss;
333 oss << std::fixed << std::setprecision(1);
334
335 if (milliseconds >= 60000) {
336 oss << static_cast<double>(milliseconds) / 60000.0 << " min";
337 } else if (milliseconds >= 1000) {
338 oss << static_cast<double>(milliseconds) / 1000.0 << " s";
339 } else {
340 oss << milliseconds << " ms";
341 }
342
343 return oss.str();
344}

346auto center_text(std::string_view text, size_t width) -> std::string {
347 if (text.length() >= width) {
348 return std::string{text.substr(0, width)};
349 }
350
351 size_t padding = (width - text.length()) / 2;
352 return std::string(padding, ' ') + std::string{text} +
353 std::string(width - padding - text.length(), ' ');
354}
355
356auto truncate(std::string_view text, size_t max_length,
357 std::string_view suffix) -> std::string {
358 if (text.length() <= max_length) {
359 return std::string{text};
360 }
361
362 if (max_length <= suffix.length()) {
363 return std::string{suffix.substr(0, max_length)};
364 }
365
366 return std::string{text.substr(0, max_length - suffix.length())} +
367 std::string{suffix};
368}
