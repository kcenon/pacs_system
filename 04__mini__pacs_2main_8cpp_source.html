37std::string generate_uid() {
38 static int counter = 0;
39 auto now = std::chrono::system_clock::now();
40 auto epoch = now.time_since_epoch();
41 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(epoch).count();
42 return "1.2.410.200001.4." + std::to_string(ms) + "." + std::to_string(++counter);
43}
