93atna_active_participant make_participant(const std::string& user_id,
94 bool is_requestor,
95 const atna_coded_value& role) {
96 atna_active_participant p;
97 p.user_id = user_id;
98 p.user_is_requestor = is_requestor;
99 p.network_access_point_type = atna_network_access_type::machine_name;
100 p.role_id_codes.push_back(role);
101 return p;
102}

104void attach_failure_detail(atna_audit_message& msg,
105 const std::string& description) {
106 if (description.empty()) return;
107 atna_object_detail d;
108 d.type = "FailureDescription";
109 d.value = description;
110 if (msg.participant_objects.empty()) {
111 atna_participant_object synth;
112 synth.object_type = atna_object_type::system_object;
113 synth.object_role = atna_object_role::report;
114 synth.object_id_type_code = kXdsDocumentUniqueIdType;
115 synth.object_id = "(no object context)";
116 synth.object_details.push_back(std::move(d));
117 msg.participant_objects.push_back(std::move(synth));
118 } else {
119 msg.participant_objects.back().object_details.push_back(std::move(d));
120 }
121}

129atna_audit_message build_iti41_audit_message(
130 const std::string& audit_source_id, const xds_iti41_event& e,
131 const std::string& local_participant_id) {
132 atna_audit_message msg;
133 msg.event_id = kEventIdExport;
134 msg.event_action = atna_event_action::create; // 'C'
135 msg.event_date_time = std::chrono::system_clock::now();
136 msg.event_outcome = e.outcome;
137 msg.event_type_codes.push_back(kEventTypeIti41);
138 msg.audit_source.audit_source_id = audit_source_id;
139
140 const std::string source =
141 local_participant_id.empty() ? e.source_id : local_participant_id;
142 msg.active_participants.push_back(
143 make_participant(source, true, atna_role_ids::source));
144 msg.active_participants.push_back(make_participant(
145 e.destination_endpoint, false, atna_role_ids::destination));
146
147 if (!e.submission_set_unique_id.empty()) {
148 atna_participant_object ss;
149 ss.object_type = atna_object_type::system_object;
150 ss.object_role = atna_object_role::report;
151 ss.object_id_type_code = kXdsSubmissionSetUidType;
152 ss.object_id = e.submission_set_unique_id;
153 msg.participant_objects.push_back(std::move(ss));
154 }
155 for (const auto& uid : e.document_unique_ids) {
156 if (uid.empty()) continue;
157 atna_participant_object d;
158 d.object_type = atna_object_type::system_object;
159 d.object_role = atna_object_role::report;
160 d.object_id_type_code = kXdsDocumentUniqueIdType;
161 d.object_id = uid;
162 msg.participant_objects.push_back(std::move(d));
163 }
164 if (!e.patient_id.empty()) {
165 atna_participant_object p;
166 p.object_type = atna_object_type::person;
167 p.object_role = atna_object_role::patient;
168 p.object_id_type_code = atna_object_id_types::patient_number;
169 p.object_id = e.patient_id;
170 msg.participant_objects.push_back(std::move(p));
171 }
172 attach_failure_detail(msg, e.failure_description);
173 return msg;
174}

176atna_audit_message build_iti43_audit_message(
177 const std::string& audit_source_id, const xds_iti43_event& e,
178 const std::string& local_participant_id) {
179 atna_audit_message msg;
180 msg.event_id = kEventIdImport;
181 msg.event_action = atna_event_action::read; // 'R'
182 msg.event_date_time = std::chrono::system_clock::now();
183 msg.event_outcome = e.outcome;
184 msg.event_type_codes.push_back(kEventTypeIti43);
185 msg.audit_source.audit_source_id = audit_source_id;
186
187 const std::string consumer =
188 local_participant_id.empty() ? e.consumer_id : local_participant_id;
189 // Consumer side: the local actor is the requestor pulling from a
190 // remote repository. Source role = remote repository (not
191 // requestor), destination role = local consumer (requestor).
192 msg.active_participants.push_back(
193 make_participant(e.source_endpoint, false, atna_role_ids::source));
194 msg.active_participants.push_back(
195 make_participant(consumer, true, atna_role_ids::destination));
196
197 if (!e.document_unique_id.empty()) {
198 atna_participant_object d;
199 d.object_type = atna_object_type::system_object;
200 d.object_role = atna_object_role::report;
201 d.object_id_type_code = kXdsDocumentUniqueIdType;
202 d.object_id = e.document_unique_id;
203 msg.participant_objects.push_back(std::move(d));
204 }
205 if (!e.repository_unique_id.empty()) {
206 atna_participant_object r;
207 r.object_type = atna_object_type::system_object;
208 r.object_role = atna_object_role::resource;
209 r.object_id_type_code = atna_object_id_types::node_id;
210 r.object_id = e.repository_unique_id;
211 msg.participant_objects.push_back(std::move(r));
212 }
213 attach_failure_detail(msg, e.failure_description);
214 return msg;
215}

217atna_audit_message build_iti18_audit_message(
218 const std::string& audit_source_id, const xds_iti18_event& e,
219 const std::string& local_participant_id) {
220 atna_audit_message msg;
221 msg.event_id = kEventIdQuery;
222 msg.event_action = atna_event_action::execute; // 'E'
223 msg.event_date_time = std::chrono::system_clock::now();
224 msg.event_outcome = e.outcome;
225 msg.event_type_codes.push_back(kEventTypeIti18);
226 msg.audit_source.audit_source_id = audit_source_id;
227
228 const std::string caller =
229 local_participant_id.empty() ? e.querier_id : local_participant_id;
230 msg.active_participants.push_back(
231 make_participant(caller, true, atna_role_ids::source));
232 msg.active_participants.push_back(make_participant(
233 e.registry_endpoint, false, atna_role_ids::destination));
234
235 if (e.query_kind == xds_iti18_event::kind::find_documents &&
236 !e.patient_id.empty()) {
237 atna_participant_object p;
238 p.object_type = atna_object_type::person;
239 p.object_role = atna_object_role::patient;
240 p.object_id_type_code = atna_object_id_types::patient_number;
241 p.object_id = e.patient_id;
242 msg.participant_objects.push_back(std::move(p));
243 }
244 for (const auto& u : e.document_uuids) {
245 if (u.empty()) continue;
246 atna_participant_object obj;
247 obj.object_type = atna_object_type::system_object;
248 obj.object_role = atna_object_role::query;
249 obj.object_id_type_code = kXdsEntryUuidType;
250 obj.object_id = u;
251 msg.participant_objects.push_back(std::move(obj));
252 }
253 attach_failure_detail(msg, e.failure_description);
254 return msg;
255}
