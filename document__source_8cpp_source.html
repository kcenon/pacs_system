32std::string make_cid(std::size_t index) {
33 static thread_local std::mt19937_64 rng(
34 static_cast<std::uint64_t>(
35 std::chrono::steady_clock::now().time_since_epoch().count()));
36 std::uniform_int_distribution<std::uint64_t> dist;
37 std::ostringstream oss;
38 oss << "doc-" << index << '-' << std::hex << dist(rng)
39 << "@kcenon.pacs";
40 return oss.str();
41}

55 kcenon::common::Result<submit_response> submit(const submission_set& s) {
56 // Collect document unique ids up-front so an audit event can
57 // reference them even when submission fails before the wire
58 // payload is built.
59 std::vector<std::string> document_uids;
60 document_uids.reserve(s.documents.size());
61 for (const auto& d : s.documents) {
62 document_uids.push_back(d.unique_id);
63 }
64
65 auto do_audit = [&](const kcenon::common::Result<submit_response>& r) {
66 audit(s, document_uids, r);
67 return r;
68 };
69
70 std::vector<std::string> cids;
71 cids.reserve(s.documents.size());
72 for (std::size_t i = 0; i < s.documents.size(); ++i) {
73 cids.push_back(make_cid(i));
74 }
75
76 auto env_result = detail::build_iti41_envelope(s, cids);
77 if (env_result.is_err()) {
78 return do_audit(kcenon::common::make_error<submit_response>(
79 env_result.error()));
80 }
81 auto env = env_result.value();
82
83 auto sign_result = detail::sign_envelope(
84 env, signing_.certificate_pem, signing_.private_key_pem,
85 signing_.private_key_password);
86 if (sign_result.is_err()) {
87 return do_audit(kcenon::common::make_error<submit_response>(
88 sign_result.error()));
89 }
90
91 std::vector<detail::mtom_part> parts;
92 parts.reserve(s.documents.size());
93 for (std::size_t i = 0; i < s.documents.size(); ++i) {
94 detail::mtom_part p;
95 p.content_id = cids[i];
96 p.mime_type = s.documents[i].mime_type;
97 p.content = s.documents[i].content;
98 parts.push_back(std::move(p));
99 }
100
101 auto mtom_result = detail::package_mtom(env.xml, parts);
102 if (mtom_result.is_err()) {
103 return do_audit(kcenon::common::make_error<submit_response>(
104 mtom_result.error()));
105 }
106 auto mtom = mtom_result.value();
107
108 auto http_result =
109 detail::http_post(opts_, mtom.content_type, mtom.body);
110 if (http_result.is_err()) {
111 return do_audit(kcenon::common::make_error<submit_response>(
112 http_result.error()));
113 }
114 auto response = http_result.value();
115
116 return do_audit(parse_registry_response(response.body,
117 s.submission_set_unique_id));
118 }
119
120private:
121 // Belt-and-suspenders cap. The HTTP layer already enforces 8 MiB via
122 // the write callback, so any response reaching us here is already
123 // bounded. This defensive check survives future refactors that might
124 // add a non-libcurl transport path without a size cap.
125 static constexpr std::size_t kMaxRegistryResponseBytes =
126 8 * 1024 * 1024;
127
128 kcenon::common::Result<submit_response> parse_registry_response(
129 const std::string& body, const std::string& submission_set_uid) {
130 if (body.size() > kMaxRegistryResponseBytes) {
131 return kcenon::common::make_error<submit_response>(
132 static_cast<int>(error_code::transport_response_too_large),
133 "RegistryResponse exceeds 8 MiB cap (" +
134 std::to_string(body.size()) + " bytes) - rejected to "
135 "bound pugixml memory footprint",
136 std::string(error_source));
137 }
138 pugi::xml_document doc;
139 auto parse = doc.load_buffer(body.data(), body.size(),
140 pugi::parse_default, pugi::encoding_utf8);
141 if (!parse) {
142 return kcenon::common::make_error<submit_response>(
143 static_cast<int>(error_code::transport_invalid_response),
144 std::string("failed to parse RegistryResponse: ") +
145 parse.description(),
146 std::string(error_source));
147 }
148
149 auto registry_response =
150 doc.select_node(
151 "//*[local-name()='RegistryResponse' or "
152 "local-name()='RegistryResponseType']")
153 .node();
154 if (!registry_response) {
155 return kcenon::common::make_error<submit_response>(
156 static_cast<int>(error_code::transport_invalid_response),
157 "response contains no RegistryResponse element",
158 std::string(error_source));
159 }
160
161 const std::string status =
162 registry_response.attribute("status").as_string();
163 if (status.empty()) {
164 return kcenon::common::make_error<submit_response>(
165 static_cast<int>(error_code::transport_invalid_response),
166 "RegistryResponse is missing status attribute",
167 std::string(error_source));
168 }
169
170 static constexpr std::string_view success_status =
171 "urn:oasis:names:tc:ebxml-regrep:ResponseStatusType:Success";
172 static constexpr std::string_view partial_success_status =
173 "urn:oasis:names:tc:ebxml-regrep:ResponseStatusType:PartialSuccess";
174
175 submit_response out;
176 out.registry_response_status = status;
177 out.submission_set_unique_id = submission_set_uid;
178
179 if (status == success_status) {
180 return out;
181 }
182 if (status == partial_success_status) {
183 return kcenon::common::make_error<submit_response>(
184 static_cast<int>(error_code::registry_partial_success),
185 "registry reported PartialSuccess: " + status,
186 std::string(error_source));
187 }
188 return kcenon::common::make_error<submit_response>(
189 static_cast<int>(error_code::registry_failure_response),
190 "registry reported non-success status: " + status,
191 std::string(error_source));
192 }

194 void audit(const submission_set& s,
195 const std::vector<std::string>& document_uids,
196 const kcenon::common::Result<submit_response>& r) {
197 if (!audit_sink_) return;
198 kcenon::pacs::security::xds_iti41_event e;
199 e.source_id = s.source_id;
200 e.destination_endpoint = opts_.endpoint;
201 e.patient_id = s.patient_id;
202 e.submission_set_unique_id = s.submission_set_unique_id;
203 e.document_unique_ids = document_uids;
204 if (r.is_ok()) {
205 e.outcome = kcenon::pacs::security::atna_event_outcome::success;
206 } else {
207 e.outcome =
208 kcenon::pacs::security::atna_event_outcome::serious_failure;
209 e.failure_description = r.error().message;
210 }
211 audit_sink_->on_iti41_submit(e);
212 }
213
214 http_options opts_;
215 signing_material signing_;
216 std::shared_ptr<kcenon::pacs::security::xds_audit_sink> audit_sink_;
217};
